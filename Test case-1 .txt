
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

import java.time.Duration;

public class ProductCategoryNavigationTest {

    public static void main(String[] args) {

        WebDriver driver = new ChromeDriver();

        try {

            // Step 1: Initialize explicit wait
            WebDriverWait wait = new WebDriverWait(
                driver, Duration.ofSeconds(10)
            );

            // Step 2: Open website
            driver.manage().window().maximize();
            driver.get("https://automationexercise.com");

            // Step 3: Verify homepage title
            String pageTitle = driver.getTitle();

            if (!pageTitle.contains("Automation Exercise")) {
                throw new AssertionError("Homepage title verification failed");
            }

            System.out.println("PASS: Homepage loaded successfully");

            // Step 4: Verify Categories section
            WebElement categories = wait.until(
                ExpectedConditions.visibilityOfElementLocated(
                    By.xpath("//*[contains(text(),'Category')]")
                )
            );

            System.out.println("PASS: Categories section is visible");

            // Step 5: Click Women category
            WebElement womenCategory = wait.until(
                ExpectedConditions.elementToBeClickable(
                    By.xpath("//a[contains(@href,'category_products/1')]")
                )
            );

            womenCategory.click();

            // Step 6: Verify navigation
            wait.until(ExpectedConditions.urlContains("category_products/1"));

            // Step 7: Verify category heading
            WebElement categoryHeading = wait.until(
                ExpectedConditions.visibilityOfElementLocated(
                    By.xpath("//*[contains(text(),'Women')]")
                )
            );

            if (categoryHeading.isDisplayed()) {
                System.out.println(
                    "PASS: Successfully navigated to Women category"
                );
            }

        } catch (Exception e) {

            System.out.println("FAIL: Test execution failed");
            e.printStackTrace();

            throw new RuntimeException(e);

        } finally {

            // Step 8: Close browser
            driver.quit();
        }
    }
}
